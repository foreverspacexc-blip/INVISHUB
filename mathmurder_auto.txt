--==========================================================
-- MATH SOLVER
--==========================================================
local function SolveMath(a, op, b)
    a = tonumber(a); b = tonumber(b)
    if not a or not b then return nil end
    op = op:gsub("%s","")
    if op == "+" then return a + b
    elseif op == "-" then return a - b
    elseif op == "x" or op == "*" or op == "X" or op == "times" or op == "x" then return a * b
    elseif op == "/" or op == "div" then
        if b == 0 then return nil end
        local r = a / b
        return math.floor(r * 1000 + 0.5) / 1000
    end
    return nil
end

-- Parse math expression from any text
local function ParseQuestion(text)
    if not text or text == "" then return nil end
    text = text:gsub("[Xx×]", "*"):gsub("[÷]", "/"):gsub("=.*$",""):gsub("%?","")
    -- Match: number op number (handles negatives and decimals)
    local a, op, b = text:match("(-?%d+%.?%d*)%s*([%+%-%*/])%s*(-?%d+%.?%d*)")
    if a and op and b then
        local ans = SolveMath(a, op, b)
        if ans then
            -- Return integer if whole number
            if ans == math.floor(ans) then return tostring(math.floor(ans)) end
            return tostring(ans)
        end
    end
    return nil
end

-- QUESTION DETECTION  (multi-layer)

local LastQuestion = ""

local _ansTextBox  = nil

-- Layer 1: scan all TextLabels in PlayerGui
local function ScanGuiForQuestion()
    for _, v in ipairs(PGui:GetDescendants()) do
        if (v:IsA("TextLabel") or v:IsA("TextButton")) and v.Visible then
            local t = v.Text
            if t and (t:find("%d") and (t:find("%+") or t:find("%-") or t:find("[Xx%*]") or t:find("/"))) then
                local ans = ParseQuestion(t)
                if ans then return ans, t, v end
            end
        end
    end
    return nil
end

-- Layer 2: watch for Attribute changes on character/player
local function WatchAttributes()
    local function Check(obj)
        if not obj then return end
        local q = obj:GetAttribute("CurrentQuestion") or obj:GetAttribute("Question") or obj:GetAttribute("MathQuestion")
        if q then
            local ans = ParseQuestion(tostring(q))
            if ans then return ans, tostring(q) end
        end
    end
    return Check(LP) or Check(LP.Character)
end

-- Layer 3: find TextBox that accepts answer
local function FindAnswerBox()
    for _, v in ipairs(PGui:GetDescendants()) do
        if v:IsA("TextBox") and v.Visible then
            local name = v.Name:lower()
            if name:find("answer") or name:find("input") or name:find("math") or
               name:find("type") or name:find("guess") or name:find("text") then
                return v
            end
        end
    end
    -- Fallback: any focused or visible textbox
    for _, v in ipairs(PGui:GetDescendants()) do
        if v:IsA("TextBox") and v.Visible and v.Parent and v.Parent.Visible then
            return v
        end
    end
    return nil
end

-- Layer 4: hook __namecall to auto-capture the answer remote
local _capturedRemotes = {}
local _mt = getrawmetatable and getrawmetatable(game)
if _mt then
    local _old = _mt.__namecall
    pcall(function()
        setreadonly(_mt, false)
        _mt.__namecall = newcclosure(function(self, ...)
            local m = getnamecallmethod()
            if (m == "FireServer" or m == "InvokeServer") and self:IsA("RemoteEvent") or self:IsA("RemoteFunction") then
                local name = self.Name:lower()
                if name:find("answer") or name:find("submit") or name:find("math") or name:find("guess") then
                    _ansRemote = self
                    _capturedRemotes[name] = { remote = self, args = {...}, method = m }
                end
            end
            return _old(self, ...)
        end)
        setreadonly(_mt, true)
    end)
end

--==========================================================
-- SUBMIT ANSWER
--==========================================================
local function SubmitAnswer(answer)
    if not answer then return false end
    local submitted = false

    -- Method A: fire captured answer remote
    if _ansRemote then
        pcall(function()
            if _ansRemote:IsA("RemoteEvent") then
                _ansRemote:FireServer(answer)
                _ansRemote:FireServer(tonumber(answer))
            else
                _ansRemote:InvokeServer(answer)
            end
            submitted = true
        end)
    end

    for _, data in pairs(_capturedRemotes) do
        pcall(function()
            if data.method == "FireServer" then
                data.remote:FireServer(answer)
                data.remote:FireServer(tonumber(answer))
            else
                data.remote:InvokeServer(answer)
            end
            submitted = true
        end)
    end

    -- Method B: search for any answer remote in RS
    if not submitted then
        for _, obj in ipairs(RS:GetDescendants()) do
            if (obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction")) then
                local n = obj.Name:lower()
                if n:find("answer") or n:find("submit") or n:find("guess") or n:find("math") then
                    pcall(function()
                        if obj:IsA("RemoteEvent") then
                            obj:FireServer(answer)
                            obj:FireServer(tonumber(answer))
                        else
                            obj:InvokeServer(answer)
                        end
                        submitted = true
                    end)
                end
            end
        end
    end

    -- Method C: type into TextBox + simulate Enter
    local box = FindAnswerBox()
    if box then
        pcall(function()
            box:CaptureFocus()
            box.Text = tostring(answer)
            -- Fire FocusLost with enterPressed = true
            local ok = pcall(function() box.FocusLost:Fire(true) end)
            if not ok then
                -- Simulate Enter key press
                if VIM then
                    VIM:SendKeyEvent(true,  Enum.KeyCode.Return, false, game)
                    VIM:SendKeyEvent(false, Enum.KeyCode.Return, false, game)
                end
            end
            submitted = true
        end)
    end

    -- Method D: simulate keyboard typing character by character
    if not submitted or box then
        pcall(function()
            local b = FindAnswerBox()
            if not b then return end
            b:CaptureFocus()
            b.Text = ""
            task.wait(0.05)
            b.Text = tostring(answer)
            task.wait(0.05)
            if VIM then
                VIM:SendKeyEvent(true,  Enum.KeyCode.Return, false, game)
                task.wait(0.03)
                VIM:SendKeyEvent(false, Enum.KeyCode.Return, false, game)
            end
        end)
    end

    return submitted
end

--==========================================================
-- AUTO ANSWER LOOP
--==========================================================
local _lastAnsweredQ = ""
local _answerCooldown = 0

task.spawn(function()
    while true do
        task.wait(0.08)
        if not Cfg.AutoAnswer then continue end

        local now = os.clock()
        if now < _answerCooldown then continue end

        -- Try all detection layers
        local answer, questionText = nil, nil

        -- Layer 1: GUI scan
        local a, qt = ScanGuiForQuestion()
        if a then answer = a; questionText = qt end

        -- Layer 2: Attributes
        if not answer then
            local a2, qt2 = WatchAttributes()
            if a2 then answer = a2; questionText = qt2 end
        end

        if answer and questionText ~= _lastAnsweredQ then
            _lastAnsweredQ = questionText
            local ok = SubmitAnswer(answer)
            if ok then
                Stats.Correct   = Stats.Correct + 1
                Stats.Streak    = Stats.Streak + 1
                Stats.CoinsEst  = Stats.CoinsEst + 10
                if Stats.Streak > Stats.BestStreak then Stats.BestStreak = Stats.Streak end
                _answerCooldown = now + 0.25
                UpdateStats()
            end
        end
    end
end)

-- SPEED BOOST (restore correct speed for game)
--==========================================================
local _speedConn = nil
local function SetSpeed(on)
    if _speedConn then _speedConn:Disconnect(); _speedConn = nil end
    if on then
        _speedConn = RunService.Heartbeat:Connect(function()
            pcall(function()
                local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
                if hum and hum.WalkSpeed ~= Cfg.SpeedValue then
                    hum.WalkSpeed = Cfg.SpeedValue
                end
            end)
        end)
    else
        pcall(function()
            local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
            if hum then hum.WalkSpeed = 16 end
        end)
    end
end

LP.CharacterAdded:Connect(function()
    task.wait(0.5)
    if Cfg.SpeedBoost then SetSpeed(true) end
end)

-- ESP COINS  (Drawing highlights on coin parts)

local ESPDrawings = {}

local function ClearESP()
    for _, d in pairs(ESPDrawings) do pcall(function() d:Remove() end) end
    ESPDrawings = {}
end

local function UpdateESP()
    if not Cfg.ESPCoins then ClearESP(); return end
    local cam = workspace.CurrentCamera
    for _, v in ipairs(workspace:GetDescendants()) do
        if v:IsA("BasePart") then
            local n = v.Name:lower()
            local par = v.Parent and v.Parent.Name:lower() or ""
            if n:find("coin") or n:find("gold") or n:find("reward") or par:find("coin") or par:find("pickup") then
                if not ESPDrawings[v] then
                    local lbl = Drawing.new("Text")
                    lbl.Size = 14; lbl.Font = 2; lbl.Center = true; lbl.Outline = true
                    lbl.Color = Color3.fromRGB(255, 215, 0); lbl.Text = "COIN"
                    lbl.Visible = true
                    ESPDrawings[v] = lbl
                end
                local sp, onS = cam:WorldToViewportPoint(v.Position)
                if onS then
                    ESPDrawings[v].Visible  = true
                    ESPDrawings[v].Position = Vector2.new(sp.X, sp.Y - 14)
                else
                    ESPDrawings[v].Visible = false
                end
            end
        end
    end
end

RunService.RenderStepped:Connect(function()
    pcall(UpdateESP)
end)

function UpdateStats()
    -- updated by GUI below
    if _statsLbl and _statsLbl.Parent then
        local elapsed = math.floor(os.clock() - Stats.StartTime)
        local mins = math.floor(elapsed / 60)

        _statsLbl.Text = string.format(

            Stats.Correct, Stats.Wrong, Stats.Streak, Stats.BestStreak,
            Stats.CoinsEst, mins, secs
        )
    end
end

    local L=Lbl(H,label,12,C_TEXT); L.Position=UDim2.fromOffset(12,6); L.Size=UDim2.new(1,-70,0,16)
    local VL=Lbl(H,tostring(val),11,C_GOLD,Enum.TextXAlignment.Right); VL.Position=UDim2.new(1,-58,0,6); VL.Size=UDim2.fromOffset(48,16)
    local TBG=Instance.new("Frame"); TBG.Size=UDim2.new(1,-24,0,5); TBG.Position=UDim2.fromOffset(12,36)
    TBG.BackgroundColor3=C_STROKE; TBG.Parent=H; Rnd(TBG,4)
    local Fill=Instance.new("Frame"); Fill.BackgroundColor3=C_GOLD; Fill.Size=UDim2.new((val-min)/(max-min),0,1,0); Fill.Parent=TBG; Rnd(Fill,4)
    local Knob=Instance.new("Frame"); Knob.Size=UDim2.fromOffset(11,11); Knob.AnchorPoint=Vector2.new(0.5,0.5)
    Knob.Position=UDim2.new((val-min)/(max-min),0,0.5,0); Knob.BackgroundColor3=C_TEXT; Knob.Parent=TBG; Rnd(Knob,20)
    local drg=false
    local function Upd(px)
        local pct=math.clamp((px-TBG.AbsolutePosition.X)/TBG.AbsoluteSize.X,0,1)
        val=math.floor(min+(max-min)*pct)
        Fill.Size=UDim2.new(pct,0,1,0); Knob.Position=UDim2.new(pct,0,0.5,0); VL.Text=tostring(val)
        if onChange then onChange(val) end
    end
    TBG.InputBegan:Connect(function(i) if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then drg=true; Upd(i.Position.X) end end)
    UIS.InputChanged:Connect(function(i) if drg and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then Upd(i.Position.X) end end)
    UIS.InputEnded:Connect(function(i) if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then drg=false end end)
end
